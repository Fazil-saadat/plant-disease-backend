package com.example.plantguard.adapters;

import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.TextView;
import androidx.annotation.NonNull;
import androidx.recyclerview.widget.RecyclerView;
import com.example.plantguard.R;
import com.example.plantguard.models.ChatMessage;
import java.util.List;

public class ChatAdapter extends RecyclerView.Adapter<RecyclerView.ViewHolder> {

    private static final int TYPE_USER = 1;
    private static final int TYPE_BOT = 2;

    private List<ChatMessage> chatMessages;

    public ChatAdapter(List<ChatMessage> chatMessages) {
        this.chatMessages = chatMessages;
    }

    @Override
    public int getItemViewType(int position) {
        return chatMessages.get(position).isUser() ? TYPE_USER : TYPE_BOT;
    }

    @NonNull
    @Override
    public RecyclerView.ViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
        if (viewType == TYPE_USER) {
            View view = LayoutInflater.from(parent.getContext()).inflate(R.layout.item_chat_user, parent, false);
            return new UserMessageViewHolder(view);
        } else {
            View view = LayoutInflater.from(parent.getContext()).inflate(R.layout.item_chat_bot, parent, false);
            return new BotMessageViewHolder(view);
        }
    }

    @Override
    public void onBindViewHolder(@NonNull RecyclerView.ViewHolder holder, int position) {
        ChatMessage message = chatMessages.get(position);

        if (holder instanceof UserMessageViewHolder) {
            ((UserMessageViewHolder) holder).bind(message);
        } else if (holder instanceof BotMessageViewHolder) {
            ((BotMessageViewHolder) holder).bind(message);
        }
    }

    @Override
    public int getItemCount() {
        return chatMessages.size();
    }

    static class UserMessageViewHolder extends RecyclerView.ViewHolder {
        TextView tvMessage;

        UserMessageViewHolder(@NonNull View itemView) {
            super(itemView);
            tvMessage = itemView.findViewById(R.id.tvUserMessage);
        }

        void bind(ChatMessage message) {
            tvMessage.setText(message.getMessage());
        }
    }

    static class BotMessageViewHolder extends RecyclerView.ViewHolder {
        TextView tvMessage;

        BotMessageViewHolder(@NonNull View itemView) {
            super(itemView);
            tvMessage = itemView.findViewById(R.id.tvBotMessage);
        }

        void bind(ChatMessage message) {
            tvMessage.setText(message.getMessage());
        }
    }
}

package com.example.plantguard.adapters;

import android.content.Context;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.Filter;
import android.widget.Filterable;
import android.widget.ImageView;
import android.widget.TextView;
import androidx.annotation.NonNull;
import androidx.recyclerview.widget.RecyclerView;
import com.example.plantguard.R;
import com.example.plantguard.models.LibraryDisease;
import java.util.ArrayList;
import java.util.List;

public class DiseaseAdapter extends RecyclerView.Adapter<DiseaseAdapter.DiseaseViewHolder> implements Filterable {

    private List<LibraryDisease> diseaseList;
    private List<LibraryDisease> diseaseListFull;
    private Context context;
    private OnItemClickListener onItemClickListener;

    public interface OnItemClickListener {
        void onItemClick(LibraryDisease disease, int position);
    }

    public void setOnItemClickListener(OnItemClickListener listener) {
        this.onItemClickListener = listener;
    }

    public DiseaseAdapter(List<LibraryDisease> diseaseList, Context context) {
        this.diseaseList = new ArrayList<>(diseaseList);
        this.diseaseListFull = new ArrayList<>(diseaseList);
        this.context = context;
    }

    @NonNull
    @Override
    public DiseaseViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
        View view = LayoutInflater.from(parent.getContext())
                .inflate(R.layout.item_disease, parent, false);
        return new DiseaseViewHolder(view);
    }

    @Override
    public void onBindViewHolder(@NonNull DiseaseViewHolder holder, int position) {
        LibraryDisease disease = diseaseList.get(position);

        if (disease != null) {
            holder.diseaseName.setText(disease.getName() != null ? disease.getName() : "Unknown Disease");
            holder.diseaseDescription.setText(disease.getDescription() != null ? disease.getDescription() : "No description available");

            // Set image
            int resourceId = context.getResources().getIdentifier(
                    "ic_plant_default", "drawable", context.getPackageName());
            if (resourceId != 0) {
                holder.diseaseImage.setImageResource(resourceId);
            } else {
                holder.diseaseImage.setImageResource(R.drawable.ic_plant_default);
            }

            // Set click listener
            holder.itemView.setOnClickListener(v -> {
                if (onItemClickListener != null) {
                    onItemClickListener.onItemClick(disease, position);
                }
            });
        }
    }

    @Override
    public int getItemCount() {
        return diseaseList != null ? diseaseList.size() : 0;
    }

    @Override
    public Filter getFilter() {
        return diseaseFilter;
    }

    private Filter diseaseFilter = new Filter() {
        @Override
        protected FilterResults performFiltering(CharSequence constraint) {
            List<LibraryDisease> filteredList = new ArrayList<>();

            if (constraint == null || constraint.length() == 0) {
                filteredList.addAll(diseaseListFull);
            } else {
                String filterPattern = constraint.toString().toLowerCase().trim();

                for (LibraryDisease disease : diseaseListFull) {
                    if (disease != null) {
                        String name = disease.getName() != null ? disease.getName().toLowerCase() : "";
                        String description = disease.getDescription() != null ? disease.getDescription().toLowerCase() : "";
                        String plantType = disease.getPlantType() != null ? disease.getPlantType().toLowerCase() : "";

                        if (name.contains(filterPattern) ||
                                description.contains(filterPattern) ||
                                plantType.contains(filterPattern)) {
                            filteredList.add(disease);
                        }
                    }
                }
            }

            FilterResults results = new FilterResults();
            results.values = filteredList;
            results.count = filteredList.size();
            return results;
        }

        @Override
        @SuppressWarnings("unchecked")
        protected void publishResults(CharSequence constraint, FilterResults results) {
            if (results.values != null) {
                diseaseList.clear();
                diseaseList.addAll((List<LibraryDisease>) results.values);
                notifyDataSetChanged();
            }
        }
    };

    public void updateFullList(List<LibraryDisease> newList) {
        if (newList != null) {
            diseaseListFull = new ArrayList<>(newList);
            diseaseList = new ArrayList<>(newList);
            notifyDataSetChanged();
        }
    }

    public List<LibraryDisease> getCurrentList() {
        return new ArrayList<>(diseaseList);
    }

    public static class DiseaseViewHolder extends RecyclerView.ViewHolder {
        ImageView diseaseImage;
        TextView diseaseName;
        TextView diseaseDescription;

        public DiseaseViewHolder(@NonNull View itemView) {
            super(itemView);
            diseaseImage = itemView.findViewById(R.id.diseaseImage);
            diseaseName = itemView.findViewById(R.id.diseaseName);
            diseaseDescription = itemView.findViewById(R.id.diseaseDescription);
        }
    }
}

import java.io.File;

public class RecentScansAdapter extends RecyclerView.Adapter<RecentScansAdapter.ScanViewHolder> {

    private List<ScanHistory> scanList;
    private OnItemClickListener onItemClickListener;

    public interface OnItemClickListener {
        void onItemClick(ScanHistory scan, int position);
    }

    public RecentScansAdapter(List<ScanHistory> scanList) {
        this.scanList = scanList;
    }

    public void setOnItemClickListener(OnItemClickListener listener) {
        this.onItemClickListener = listener;
    }

    @NonNull
    @Override
    public ScanViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
        View view = LayoutInflater.from(parent.getContext())
                .inflate(R.layout.item_scan_history, parent, false);
        return new ScanViewHolder(view);
    }

    @Override
    public void onBindViewHolder(@NonNull ScanViewHolder holder, int position) {
        ScanHistory scan = scanList.get(position);

        holder.diseaseName.setText(scan.getDiseaseName());
        holder.confidence.setText(String.format("%.1f%%", scan.getConfidence()));
        holder.date.setText(scan.getScanDate());

        // Load image from file path
        if (scan.getImagePath() != null && !scan.getImagePath().isEmpty()) {
            File imageFile = new File(scan.getImagePath());
            if (imageFile.exists()) {
                Bitmap bitmap = BitmapFactory.decodeFile(imageFile.getAbsolutePath());
                if (bitmap != null) {
                    holder.plantImage.setImageBitmap(bitmap);
                } else {
                    holder.plantImage.setImageResource(R.drawable.ic_plant_default);
                }
            } else {
                holder.plantImage.setImageResource(R.drawable.ic_plant_default);
            }
        } else {
            holder.plantImage.setImageResource(R.drawable.ic_plant_default);
        }

        // Set confidence color
        if (scan.getConfidence() > 80) {
            holder.confidence.setTextColor(ContextCompat.getColor(holder.itemView.getContext(), android.R.color.holo_green_dark));
        } else if (scan.getConfidence() > 60) {
            holder.confidence.setTextColor(ContextCompat.getColor(holder.itemView.getContext(), android.R.color.holo_orange_dark));
        } else {
            holder.confidence.setTextColor(ContextCompat.getColor(holder.itemView.getContext(), android.R.color.holo_red_dark));
        }

        // Set click listener with position
        holder.itemView.setOnClickListener(v -> {
            if (onItemClickListener != null) {
                onItemClickListener.onItemClick(scan, position);
            }
        });
    }

    @Override
    public int getItemCount() {
        return scanList.size();
    }

    public static class ScanViewHolder extends RecyclerView.ViewHolder {
        ImageView plantImage;
        TextView diseaseName;
        TextView confidence;
        TextView date;

        public ScanViewHolder(@NonNull View itemView) {
            super(itemView);
            plantImage = itemView.findViewById(R.id.plantImage);
            diseaseName = itemView.findViewById(R.id.diseaseName);
            confidence = itemView.findViewById(R.id.confidence);
            date = itemView.findViewById(R.id.date);
        }
    }
}

package com.example.plantguard.fragments;

import android.content.Intent;
import android.graphics.Bitmap;
import android.graphics.BitmapFactory;
import android.net.Uri;
import android.os.Bundle;
import android.os.Environment;
import android.provider.MediaStore;
import android.util.Log;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.Toast;
import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.core.content.FileProvider;
import androidx.fragment.app.Fragment;
import androidx.navigation.Navigation;

import com.example.plantguard.R;
import com.example.plantguard.databinding.FragmentCameraBinding;
import java.io.File;
import java.io.IOException;
import java.text.SimpleDateFormat;
import java.util.Date;
import java.util.Locale;

public class CameraFragment extends Fragment {

    private FragmentCameraBinding binding;
    private static final int REQUEST_IMAGE_CAPTURE = 1002;
    private static final int REQUEST_IMAGE_PICK = 1003;
    private Uri selectedImageUri;
    private String currentPhotoPath;
    private static final String TAG = "CameraFragment";

    @Nullable
    @Override
    public View onCreateView(@NonNull LayoutInflater inflater, @Nullable ViewGroup container, @Nullable Bundle savedInstanceState) {
        binding = FragmentCameraBinding.inflate(inflater, container, false);
        return binding.getRoot();
    }

    @Override
    public void onViewCreated(@NonNull View view, @Nullable Bundle savedInstanceState) {
        super.onViewCreated(view, savedInstanceState);

        // Check if we have a pre-selected image from HomeFragment
        Bundle args = getArguments();
        if (args != null) {
            if (args.containsKey("image_path")) {
                // Image from camera
                currentPhotoPath = args.getString("image_path");
                File file = new File(currentPhotoPath);
                if (file.exists()) {
                    selectedImageUri = Uri.fromFile(file);
                    displayHighQualityImage(selectedImageUri);
                    binding.btnAnalyze.setEnabled(true);
                    Toast.makeText(requireContext(), "Image captured successfully", Toast.LENGTH_SHORT).show();
                }
            } else if (args.containsKey("image_uri")) {
                // Image from gallery
                String imageUriString = args.getString("image_uri");
                selectedImageUri = Uri.parse(imageUriString);
                binding.imagePreview.setImageURI(selectedImageUri);
                binding.btnAnalyze.setEnabled(true);
                Toast.makeText(requireContext(), "Image selected from gallery", Toast.LENGTH_SHORT).show();
            }
        }

        setupClickListeners();
    }

    private void setupClickListeners() {
        binding.btnCapture.setOnClickListener(v -> openCamera());
        binding.btnGallery.setOnClickListener(v -> openGallery());
        binding.btnAnalyze.setOnClickListener(v -> analyzeImage());
    }

    private void openCamera() {
        Intent takePictureIntent = new Intent(MediaStore.ACTION_IMAGE_CAPTURE);

        if (takePictureIntent.resolveActivity(requireActivity().getPackageManager()) != null) {
            File photoFile = null;
            try {
                photoFile = createImageFile();
            } catch (IOException ex) {
                Log.e(TAG, "Error creating image file", ex);
                Toast.makeText(requireContext(), "Error creating image file", Toast.LENGTH_SHORT).show();
                return;
            }

            if (photoFile != null) {
                Uri photoURI = FileProvider.getUriForFile(requireContext(),
                        "com.example.plantguard.fileprovider",
                        photoFile);

                takePictureIntent.putExtra(MediaStore.EXTRA_OUTPUT, photoURI);
                startActivityForResult(takePictureIntent, REQUEST_IMAGE_CAPTURE);
            }
        } else {
            Toast.makeText(requireContext(), "No camera app available", Toast.LENGTH_SHORT).show();
        }
    }

    private File createImageFile() throws IOException {
        String timeStamp = new SimpleDateFormat("yyyyMMdd_HHmmss", Locale.getDefault()).format(new Date());
        String imageFileName = "PLANT_" + timeStamp;
        File storageDir = requireContext().getExternalFilesDir(Environment.DIRECTORY_PICTURES);

        if (!storageDir.exists()) {
            storageDir.mkdirs();
        }

        File imageFile = File.createTempFile(
                imageFileName,
                ".jpg",
                storageDir
        );

        currentPhotoPath = imageFile.getAbsolutePath();
        return imageFile;
    }

    private void openGallery() {
        Intent intent = new Intent(Intent.ACTION_PICK, MediaStore.Images.Media.EXTERNAL_CONTENT_URI);
        intent.setType("image/*");
        startActivityForResult(intent, REQUEST_IMAGE_PICK);
    }

    private void analyzeImage() {
        if (selectedImageUri != null) {
            openResultFragment(selectedImageUri);
        } else {
            Toast.makeText(requireContext(), "Please select an image first", Toast.LENGTH_SHORT).show();
        }
    }

    @Override
    public void onActivityResult(int requestCode, int resultCode, @Nullable Intent data) {
        super.onActivityResult(requestCode, resultCode, data);

        if (resultCode == getActivity().RESULT_OK) {
            if (requestCode == REQUEST_IMAGE_CAPTURE) {
                // Camera result - image is saved to file
                if (currentPhotoPath != null) {
                    File file = new File(currentPhotoPath);
                    if (file.exists()) {
                        selectedImageUri = Uri.fromFile(file);
                        displayHighQualityImage(selectedImageUri);
                        binding.btnAnalyze.setEnabled(true);
                        Toast.makeText(requireContext(), "Image captured successfully", Toast.LENGTH_SHORT).show();
                    }
                }
            } else if (requestCode == REQUEST_IMAGE_PICK && data != null) {
                // Gallery pick
                selectedImageUri = data.getData();
                if (selectedImageUri != null) {
                    binding.imagePreview.setImageURI(selectedImageUri);
                    binding.btnAnalyze.setEnabled(true);
                    Toast.makeText(requireContext(), "Image selected from gallery", Toast.LENGTH_SHORT).show();
                }
            }
        }
    }

    private void displayHighQualityImage(Uri imageUri) {
        try {
            if (imageUri.getScheme().equals("file")) {
                // For file URIs
                BitmapFactory.Options options = new BitmapFactory.Options();
                options.inSampleSize = 1;
                Bitmap bitmap = BitmapFactory.decodeFile(imageUri.getPath(), options);
                if (bitmap != null) {
                    binding.imagePreview.setImageBitmap(bitmap);
                }
            } else {
                // For content URIs
                binding.imagePreview.setImageURI(imageUri);
            }
        } catch (Exception e) {
            Log.e(TAG, "Error displaying image", e);
            binding.imagePreview.setImageURI(imageUri);
        }
    }

    private void openResultFragment(Uri imageUri) {
        try {
            Bundle args = new Bundle();
            args.putString("image_uri", imageUri.toString());

            if (currentPhotoPath != null && imageUri.getScheme().equals("file")) {
                args.putString("image_path", currentPhotoPath);
            }

            Navigation.findNavController(requireView()).navigate(R.id.action_cameraFragment_to_resultFragment, args);
        } catch (Exception e) {
            Log.e(TAG, "Error opening result fragment: " + e.getMessage(), e);
            Toast.makeText(requireContext(), "Error opening analysis", Toast.LENGTH_SHORT).show();
        }
    }

    @Override
    public void onDestroyView() {
        super.onDestroyView();
        binding = null;
    }
}

package com.example.plantguard.fragments;

import android.os.Bundle;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.Toast;
import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.fragment.app.Fragment;
import androidx.recyclerview.widget.LinearLayoutManager;
import com.example.plantguard.databinding.FragmentChatbotBinding;
import com.example.plantguard.adapters.ChatAdapter;
import com.example.plantguard.models.ChatMessage;
import java.util.ArrayList;
import java.util.List;
import java.util.Random;

public class ChatBotFragment extends Fragment {

    private FragmentChatbotBinding binding;
    private ChatAdapter chatAdapter;
    private List<ChatMessage> chatMessages;

    // Plant disease knowledge base
    private final String[][] plantKnowledge = {
            // Common plant diseases and their solutions
            {"yellow leaves", "Yellow leaves can indicate overwatering, nutrient deficiency, or lack of sunlight. Check soil moisture and ensure proper drainage."},
            {"brown spots", "Brown spots may be caused by fungal infections, sunburn, or bacterial diseases. Remove affected leaves and apply appropriate fungicide."},
            {"white powder", "White powder is likely powdery mildew. Improve air circulation and apply sulfur-based fungicide."},
            {"wilting", "Wilting can be due to underwatering, root rot, or pests. Check soil moisture and inspect roots for rot."},
            {"leaf curl", "Leaf curl can be caused by pests, viruses, or environmental stress. Inspect for insects and ensure proper watering."},
            {"black spots", "Black spots are often black spot fungus. Remove infected leaves and apply fungicide regularly."},
            {"mold", "Mold indicates too much moisture. Reduce watering and improve ventilation around the plant."}
    };

    // General plant care tips
    private final String[] generalTips = {
            "Water plants when the top inch of soil feels dry.",
            "Ensure plants get adequate sunlight based on their species requirements.",
            "Use well-draining soil to prevent root rot.",
            "Fertilize during growing season for optimal growth.",
            "Regularly inspect plants for pests and diseases.",
            "Prune dead or yellow leaves to encourage new growth.",
            "Rotate plants periodically for even growth."
    };

    @Nullable
    @Override
    public View onCreateView(@NonNull LayoutInflater inflater, @Nullable ViewGroup container, @Nullable Bundle savedInstanceState) {
        binding = FragmentChatbotBinding.inflate(inflater, container, false);
        return binding.getRoot();
    }

    @Override
    public void onViewCreated(@NonNull View view, @Nullable Bundle savedInstanceState) {
        super.onViewCreated(view, savedInstanceState);

        setupChat();
        setupClickListeners();
        sendWelcomeMessage();
    }

    private void setupChat() {
        chatMessages = new ArrayList<>();
        chatAdapter = new ChatAdapter(chatMessages);

        binding.chatRecyclerView.setLayoutManager(new LinearLayoutManager(requireContext()));
        binding.chatRecyclerView.setAdapter(chatAdapter);

        // Smooth scrolling to bottom
        binding.chatRecyclerView.scrollToPosition(chatMessages.size() - 1);
    }

    private void setupClickListeners() {
        binding.btnSend.setOnClickListener(v -> sendMessage());

        // Send message when user presses enter
        binding.etMessage.setOnClickListener(v -> sendMessage());
    }

    private void sendWelcomeMessage() {
        String welcomeMessage = "👋 Hello! I'm your PlantGuard AI assistant. I can help you with:\n\n" +
                "• Plant disease identification\n" +
                "• Plant care tips\n" +
                "• Treatment recommendations\n" +
                "• General gardening advice\n\n" +
                "How can I help you today?";

        addMessage(welcomeMessage, false);
    }

    private void sendMessage() {
        String message = binding.etMessage.getText().toString().trim();

        if (!message.isEmpty()) {
            // Add user message
            addMessage(message, true);
            binding.etMessage.setText("");

            // Generate and send bot response
            generateBotResponse(message);
        }
    }

    private void generateBotResponse(String userMessage) {
        String lowerMessage = userMessage.toLowerCase();
        String response;

        // Check for specific plant disease queries
        boolean foundDisease = false;
        for (String[] knowledge : plantKnowledge) {
            if (lowerMessage.contains(knowledge[0])) {
                response = "🌿 Based on your description of \"" + knowledge[0] + "\":\n\n" + knowledge[1] +
                        "\n\n💡 Recommendation: Monitor the plant closely and consider consulting with a local nursery for specific treatment options.";
                addMessage(response, false);
                foundDisease = true;
                break;
            }
        }

        if (!foundDisease) {
            // Check for general plant care queries
            if (lowerMessage.contains("water") || lowerMessage.contains("watering")) {
                response = "💧 Watering Tips:\n\n" + generalTips[0] + "\n\nDifferent plants have different water needs. " +
                        "Succulents need less water, while tropical plants need more consistent moisture.";
            }
            else if (lowerMessage.contains("sun") || lowerMessage.contains("light")) {
                response = "☀️ Light Requirements:\n\n" + generalTips[1] + "\n\nMost plants need 6-8 hours of indirect sunlight daily. " +
                        "Check your specific plant's light requirements.";
            }
            else if (lowerMessage.contains("soil") || lowerMessage.contains("potting")) {
                response = "🪴 Soil Advice:\n\n" + generalTips[2] + "\n\nUse potting mix specific to your plant type. " +
                        "Cactus mix for succulents, orchid mix for orchids, etc.";
            }
            else if (lowerMessage.contains("fertiliz") || lowerMessage.contains("food")) {
                response = "🌱 Fertilizing Guide:\n\n" + generalTips[3] + "\n\nUse balanced fertilizer during spring and summer. " +
                        "Reduce or stop fertilizing in winter when plants are dormant.";
            }
            else if (lowerMessage.contains("hello") || lowerMessage.contains("hi") || lowerMessage.contains("hey")) {
                response = "Hello! 👋 I'm here to help with your plant care questions. What would you like to know?";
            }
            else if (lowerMessage.contains("thank")) {
                response = "You're welcome! 😊 Feel free to ask if you have any more plant-related questions.";
            }
            else {
                // Random general tip for unrelated queries
                Random random = new Random();
                String randomTip = generalTips[random.nextInt(generalTips.length)];
                response = "I'm not sure I understand your question about plant care. Here's a general tip:\n\n" +
                        randomTip + "\n\nYou can ask me about specific plant problems, watering, lighting, or soil issues.";
            }

            addMessage(response, false);
        }
    }

    private void addMessage(String message, boolean isUser) {
        ChatMessage chatMessage = new ChatMessage(message, isUser);
        chatMessages.add(chatMessage);
        chatAdapter.notifyItemInserted(chatMessages.size() - 1);

        // Scroll to bottom
        binding.chatRecyclerView.scrollToPosition(chatMessages.size() - 1);
    }

    @Override
    public void onDestroyView() {
        super.onDestroyView();
        binding = null;
    }
}

    private void setupRecentScans() {
        // Get recent scans from SharedPreferences
        List<ScanHistory> recentScans = scanHistoryManager.getRecentScans();

        if (recentScans.isEmpty()) {
            binding.recentScansRecyclerView.setVisibility(View.GONE);
        } else {
            binding.recentScansRecyclerView.setVisibility(View.VISIBLE);
            recentScansAdapter = new RecentScansAdapter(recentScans);
            binding.recentScansRecyclerView.setLayoutManager(new LinearLayoutManager(requireContext()));
            binding.recentScansRecyclerView.setAdapter(recentScansAdapter);

            // Set click listener for scan items - pass position
            recentScansAdapter.setOnItemClickListener((scan, position) -> {
                // For recent scans, we need to find the position in the full list
                List<ScanHistory> fullHistory = scanHistoryManager.getScanHistory();
                int fullPosition = findScanPositionInFullList(scan, fullHistory);

                Bundle bundle = new Bundle();
                bundle.putSerializable("scan_history", scan);
                bundle.putInt("scan_position", fullPosition);
                Navigation.findNavController(requireView()).navigate(R.id.action_homeFragment_to_scanDetailFragment, bundle);
            });
        }
    }

    // Helper method to find scan position in full list
    private int findScanPositionInFullList(ScanHistory scan, List<ScanHistory> fullList) {
        for (int i = 0; i < fullList.size(); i++) {
            ScanHistory currentScan = fullList.get(i);
            if (currentScan.getDiseaseName().equals(scan.getDiseaseName()) &&
                    currentScan.getConfidence() == scan.getConfidence() &&
                    currentScan.getScanDate().equals(scan.getScanDate()) &&
                    currentScan.getImagePath().equals(scan.getImagePath())) {
                return i;
            }
        }
        return -1;
    }

    private void openCamera() {
        Intent takePictureIntent = new Intent(MediaStore.ACTION_IMAGE_CAPTURE);

        if (takePictureIntent.resolveActivity(requireActivity().getPackageManager()) != null) {
            File photoFile = null;
            try {
                photoFile = createImageFile();
            } catch (IOException ex) {
                Toast.makeText(requireContext(), "Error creating image file", Toast.LENGTH_SHORT).show();
                return;
            }

            if (photoFile != null) {
                Uri photoURI = FileProvider.getUriForFile(requireContext(),
                        "com.example.plantguard.fileprovider",
                        photoFile);

                takePictureIntent.putExtra(MediaStore.EXTRA_OUTPUT, photoURI);
                startActivityForResult(takePictureIntent, REQUEST_IMAGE_CAPTURE);
            }
        } else {
            Toast.makeText(requireContext(), "No camera app available", Toast.LENGTH_SHORT).show();
        }
    }

    private void openGallery() {
        Intent intent = new Intent(Intent.ACTION_PICK, MediaStore.Images.Media.EXTERNAL_CONTENT_URI);
        intent.setType("image/*");
        startActivityForResult(intent, REQUEST_IMAGE_PICK);
    }

    private File createImageFile() throws IOException {
        // Create image file with timestamp
        String timeStamp = new SimpleDateFormat("yyyyMMdd_HHmmss", Locale.getDefault()).format(new Date());
        String imageFileName = "PLANT_" + timeStamp;
        File storageDir = requireContext().getExternalFilesDir(android.os.Environment.DIRECTORY_PICTURES);

        // Create directory if it doesn't exist
        if (!storageDir.exists()) {
            storageDir.mkdirs();
        }

        File imageFile = File.createTempFile(
                imageFileName,
                ".jpg",
                storageDir
        );

        currentPhotoPath = imageFile.getAbsolutePath();
        return imageFile;
    }

    @Override
    public void onActivityResult(int requestCode, int resultCode, @Nullable Intent data) {
        super.onActivityResult(requestCode, resultCode, data);

        if (resultCode == getActivity().RESULT_OK) {
            if (requestCode == REQUEST_IMAGE_CAPTURE) {
                // Camera result - navigate to CameraFragment with the captured image
                if (currentPhotoPath != null) {
                    navigateToCameraFragmentWithImage(currentPhotoPath, true);
                }
            } else if (requestCode == REQUEST_IMAGE_PICK && data != null) {
                // Gallery pick - navigate to CameraFragment with the selected image
                Uri selectedImageUri = data.getData();
                if (selectedImageUri != null) {
                    navigateToCameraFragmentWithImage(selectedImageUri.toString(), false);
                }
            }
        }
    }

    private void navigateToCameraFragmentWithImage(String imagePath, boolean isFromCamera) {
        // Navigate to CameraFragment and pass the image data
        Bundle bundle = new Bundle();
        if (isFromCamera) {
            bundle.putString("image_path", imagePath);
            bundle.putBoolean("from_camera", true);
        } else {
            bundle.putString("image_uri", imagePath);
            bundle.putBoolean("from_gallery", true);
        }

        Navigation.findNavController(requireView()).navigate(R.id.action_homeFragment_to_cameraFragment, bundle);
    }

        super.onResume();
        // Refresh recent scans when fragment resumes
        setupRecentScans();
    }

    @Override
    public void onDestroyView() {
        super.onDestroyView();
        binding = null;
    }

public class LibraryFragment extends Fragment {

    private RecyclerView recyclerViewDiseases;

    private ProgressBar progressBar;
    private DiseaseAdapter diseaseAdapter;
    private ApiService apiService;
    private LanguageManager languageManager;
    private List<LibraryDisease> diseaseList = new ArrayList<>();

    // Empty state views
    private View emptyState;
    private ImageView emptyStateIcon;
    private TextView emptyStateTitle;
    private TextView emptyStateMessage;

    // Add this to track network call
    private Call<LibraryDiseasesResponse> currentCall;

    @Nullable
    @Override
    public View onCreateView(@NonNull LayoutInflater inflater, @Nullable ViewGroup container, @Nullable Bundle savedInstanceState) {
        View view = inflater.inflate(R.layout.fragment_library, container, false);

        super.onViewCreated(view, savedInstanceState);

        apiService = RetrofitClient.getApiService();
        languageManager = new LanguageManager(requireContext());

        initViews(view);
        fetchLibraryDiseases();
    }

    private void initViews(View view) {
        recyclerViewDiseases = view.findViewById(R.id.recyclerViewDiseases);
        searchDisease = view.findViewById(R.id.searchDisease);
        progressBar = view.findViewById(R.id.progressBar);

        // Initialize empty state views

        emptyStateIcon = view.findViewById(R.id.emptyStateIcon);

        emptyStateMessage = view.findViewById(R.id.emptyStateMessage);

        // Initialize adapter with LibraryDisease
        diseaseAdapter = new DiseaseAdapter(diseaseList, requireContext());
        recyclerViewDiseases.setLayoutManager(new LinearLayoutManager(requireContext()));
        recyclerViewDiseases.setAdapter(diseaseAdapter);

        // Setup retry button
        btnRetry.setOnClickListener(v -> fetchLibraryDiseases());

        // Setup item click listener
        diseaseAdapter.setOnItemClickListener((disease, position) -> {
            Bundle bundle = new Bundle();
            bundle.putSerializable("disease", disease);
            Navigation.findNavController(requireView()).navigate(R.id.action_libraryFragment_to_diseaseDetailFragment, bundle);
        });

        // Setup search functionality
        setupSearchFunctionality();
    }

    private void fetchLibraryDiseases() {
        // Cancel previous call if exists
        if (currentCall != null) {
            currentCall.cancel();
        }

        showLoadingState();
        String currentLanguage = languageManager.getSavedLanguage();

        currentCall = apiService.getLibraryDiseases(currentLanguage);
        currentCall.enqueue(new Callback<LibraryDiseasesResponse>() {
            @Override
            public void onResponse(Call<LibraryDiseasesResponse> call, Response<LibraryDiseasesResponse> response) {
                // Check if fragment is still attached
                if (!isAdded() || getContext() == null) {
                    return;
                }

                if (response.isSuccessful() && response.body() != null) {
                    LibraryDiseasesResponse libraryResponse = response.body();
                    if ("success".equals(libraryResponse.getStatus())) {
                        if (libraryResponse.getDiseases() != null && !libraryResponse.getDiseases().isEmpty()) {
                            updateDiseaseList(libraryResponse.getDiseases());
                            showContentState();
                            Toast.makeText(requireContext(),
                                    getString(R.string.loaded_diseases, libraryResponse.getCount()),
                                    Toast.LENGTH_SHORT).show();
                        } else {
                            showEmptyState(getString(R.string.no_diseases_available),
                                    getString(R.string.no_diseases_message));
                        }
                    } else {
                        showErrorState(getString(R.string.failed_to_load_diseases),
                                getString(R.string.disease_load_error));
                        Toast.makeText(requireContext(),
                                getString(R.string.failed_load_diseases_server),
                                Toast.LENGTH_SHORT).show();
                    }
                } else {
                    showErrorState(getString(R.string.network_error),
                            getString(R.string.network_error_message));
                    Toast.makeText(requireContext(),
                            getString(R.string.network_error_with_message, response.message()),
                            Toast.LENGTH_LONG).show();
                }

                currentCall = null; // Clear the call reference
            }

            @Override
            public void onFailure(Call<LibraryDiseasesResponse> call, Throwable t) {
                // Check if fragment is still attached and call wasn't cancelled
                if (!isAdded() || getContext() == null || call.isCanceled()) {
                    return;
                }

                showErrorState(getString(R.string.connection_failed),
                        getString(R.string.connection_failed_message));
                Toast.makeText(requireContext(),
                        getString(R.string.network_error_with_message, t.getMessage()),
                        Toast.LENGTH_LONG).show();

                currentCall = null; // Clear the call reference
            }
        });
    }

    private void updateDiseaseList(List<LibraryDisease> libraryDiseases) {
        diseaseList.clear();
        diseaseList.addAll(libraryDiseases);
        diseaseAdapter.updateFullList(diseaseList);
    }

    private void setupSearchFunctionality() {
        searchDisease.addTextChangedListener(new TextWatcher() {
            @Override
            public void beforeTextChanged(CharSequence s, int start, int count, int after) {}

            @Override
            public void onTextChanged(CharSequence s, int start, int before, int count) {
                // Filter the list based on search text
                diseaseAdapter.getFilter().filter(s);

                // Update empty state based on search results
                if (s.length() > 0 && diseaseAdapter.getItemCount() == 0) {
                    showEmptyState(getString(R.string.no_results_found),
                            getString(R.string.no_results_message, s.toString()));
                } else if (s.length() == 0 && diseaseAdapter.getItemCount() == 0) {
                    showEmptyState(getString(R.string.no_diseases_available),
                            getString(R.string.no_diseases_message));
                } else if (diseaseAdapter.getItemCount() > 0) {
                    showContentState();
                }
            }

            @Override
            public void afterTextChanged(Editable s) {}
        });
    }

    private void showLoadingState() {
        progressBar.setVisibility(View.VISIBLE);
        recyclerViewDiseases.setVisibility(View.GONE);
        emptyState.setVisibility(View.GONE);
        btnRetry.setVisibility(View.GONE);
    }

    private void showContentState() {
        progressBar.setVisibility(View.GONE);
        recyclerViewDiseases.setVisibility(View.VISIBLE);
        emptyState.setVisibility(View.GONE);
        btnRetry.setVisibility(View.GONE);
    }

    private void showEmptyState(String title, String message) {
        progressBar.setVisibility(View.GONE);
        recyclerViewDiseases.setVisibility(View.GONE);
        emptyState.setVisibility(View.VISIBLE);

        emptyStateTitle.setText(title);
        emptyStateMessage.setText(message);
        emptyStateIcon.setImageResource(R.drawable.ic_search);
        btnRetry.setVisibility(View.GONE);
    }

    private void showErrorState(String title, String message) {
        progressBar.setVisibility(View.GONE);
        recyclerViewDiseases.setVisibility(View.GONE);
        emptyState.setVisibility(View.VISIBLE);

        emptyStateTitle.setText(title);
        emptyStateMessage.setText(message);
        emptyStateIcon.setImageResource(android.R.drawable.ic_dialog_alert);
        btnRetry.setVisibility(View.VISIBLE);
    }

    @Override
    public void onDestroyView() {
        super.onDestroyView();
        // Cancel network call when fragment is destroyed
        if (currentCall != null) {
            currentCall.cancel();
            currentCall = null;
        }
    }

    private void analyzeImageWithAPI(Uri imageUri) {
        if (!isAdded() || getContext() == null) {
            return;
        }
        
        progressBar.setVisibility(View.VISIBLE);
        resultContainer.setVisibility(View.GONE);

        try {
            File file = createFileFromUri(imageUri);
            if (file == null || !file.exists()) {
                Toast.makeText(requireContext(), getString(R.string.error_processing_image, "Image file not found"), Toast.LENGTH_SHORT).show();
                progressBar.setVisibility(View.GONE);
                return;
            }

            RequestBody requestFile = RequestBody.create(MediaType.parse("image/*"), file);
            MultipartBody.Part imagePart = MultipartBody.Part.createFormData("file", file.getName(), requestFile);

            // Get current language and convert to RequestBody
            String currentLanguage = languageManager.getSavedLanguage();
            RequestBody languageBody = RequestBody.create(MediaType.parse("text/plain"), currentLanguage);

            Call<PredictionResponse> call = apiService.predictDisease(imagePart, languageBody);
            call.enqueue(new Callback<PredictionResponse>() {
                @Override
                public void onResponse(Call<PredictionResponse> call, Response<PredictionResponse> response) {
                    progressBar.setVisibility(View.GONE);

                    if (response.isSuccessful() && response.body() != null) {
                        currentPrediction = response.body(); // Store the prediction for saving
                        resultContainer.setVisibility(View.VISIBLE);
                        displayResults(currentPrediction);
                    } else {
                        Toast.makeText(requireContext(), 
                                getString(R.string.analysis_failed, response.message()), 
                                Toast.LENGTH_LONG).show();
                        Navigation.findNavController(requireView()).popBackStack();
                    }
                }

                @Override
                public void onFailure(Call<PredictionResponse> call, Throwable t) {
                    progressBar.setVisibility(View.GONE);
                    Toast.makeText(requireContext(), 
                            getString(R.string.network_error_with_message, t.getMessage()), 
                            Toast.LENGTH_LONG).show();
                    Navigation.findNavController(requireView()).popBackStack();
                }
            });

        } catch (Exception e) {
            progressBar.setVisibility(View.GONE);
            Toast.makeText(requireContext(), 
                    getString(R.string.error_processing_image, e.getMessage()), 
                    Toast.LENGTH_SHORT).show();
            Log.e("ResultFragment", "Error in analyzeImageWithAPI", e);
        }
    }

    private File createFileFromUri(Uri uri) {
        try {
            InputStream inputStream = requireContext().getContentResolver().openInputStream(uri);
            if (inputStream == null) {
                return null;
            }

            // Create a temporary file in cache directory
            File file = new File(requireContext().getCacheDir(), "plant_image_" + System.currentTimeMillis() + ".jpg");
            FileOutputStream outputStream = new FileOutputStream(file);

            byte[] buffer = new byte[4096];
            int bytesRead;
            while ((bytesRead = inputStream.read(buffer)) != -1) {
                outputStream.write(buffer, 0, bytesRead);
            }

            inputStream.close();
            outputStream.close();
            return file;

        } catch (Exception e) {
            Log.e("ResultFragment", "Error creating file from URI", e);
            return null;
        }
    }

    // Create local copy for gallery images
    private String createLocalCopyFromUri(Uri uri) {
        try {
            File storageDir = requireContext().getExternalFilesDir(android.os.Environment.DIRECTORY_PICTURES);
            if (!storageDir.exists()) {
                storageDir.mkdirs();
            }

            String timeStamp = new SimpleDateFormat("yyyyMMdd_HHmmss", Locale.getDefault()).format(new Date());
            String imageFileName = "GALLERY_" + timeStamp + ".jpg";
            File imageFile = new File(storageDir, imageFileName);

            InputStream inputStream = requireContext().getContentResolver().openInputStream(uri);
            FileOutputStream outputStream = new FileOutputStream(imageFile);

            byte[] buffer = new byte[4096];
            int bytesRead;
            while ((bytesRead = inputStream.read(buffer)) != -1) {
                outputStream.write(buffer, 0, bytesRead);
            }

            inputStream.close();
            outputStream.close();
            return imageFile.getAbsolutePath();

        } catch (Exception e) {
            Log.e("ResultFragment", "Error creating local copy from URI", e);
            return "";
        }
    }

    private void saveScanResult() {
        try {
            if (currentPrediction == null) {
                Toast.makeText(requireContext(), "No analysis result to save", Toast.LENGTH_SHORT).show();
                return;
            }

            // Get current date and time
            String currentDate = new SimpleDateFormat("dd MMM yyyy, HH:mm", Locale.getDefault()).format(new Date());

            // Use the stored image path
            String imagePath = currentImagePath;
            if (imagePath == null || imagePath.isEmpty()) {
                // Fallback: try to get from arguments
                Bundle args = getArguments();
                if (args != null) {
                    imagePath = args.getString("image_path", "");
                }
            }

            // Use the direct confidence value from prediction (fix for language issues)
            double confidenceValue = currentPrediction.getConfidence();

            // Create scan history object
            ScanHistory scan = new ScanHistory(
                    currentPrediction.getDiseaseName(), // Use the disease name from API
                    confidenceValue, // Use direct value, not parsed from text
                    currentDate,
                    imagePath != null ? imagePath : ""
            );
            scan.setTreatment(currentPrediction.getTreatment());

            // Save to SharedPreferences
            scanHistoryManager.addScanHistory(scan);

            Toast.makeText(requireContext(), getString(R.string.scan_saved_success), Toast.LENGTH_SHORT).show();
            btnSaveResult.setEnabled(false);
            btnSaveResult.setText(getString(R.string.saved));

        } catch (Exception e) {
            Toast.makeText(requireContext(), 
                    getString(R.string.error_saving_result, e.getMessage()), 
                    Toast.LENGTH_SHORT).show();
            Log.e("ResultFragment", "Error saving scan result", e);
        }
    }

    private void displayResults(PredictionResponse result) {
        // Use the localized disease_name from backend
        diseaseName.setText(result.getDiseaseName());
        
        // Store the confidence value directly for saving
        double confidence = result.getConfidence();
        
        // Format the confidence text - always use English format for numbers to avoid parsing issues
        String confidenceText = String.format(Locale.ENGLISH, "%.1f%%", confidence);
        confidenceScore.setText(confidenceText);

        // Set all the localized fields from the backend response
        diseaseDescription.setText(result.getDescription());
        symptomsList.setText(result.getSymptoms());
        treatmentsList.setText(result.getTreatment());
        preventionList.setText(result.getPrevention());

        // Confidence color coding
        if (confidence > 80) {
            confidenceScore.setTextColor(ContextCompat.getColor(requireContext(), android.R.color.holo_green_dark));
        } else if (confidence > 60) {
            confidenceScore.setTextColor(ContextCompat.getColor(requireContext(), android.R.color.holo_orange_dark));
        } else {
            confidenceScore.setTextColor(ContextCompat.getColor(requireContext(), android.R.color.holo_red_dark));
            String lowConfidenceText = getString(R.string.low_confidence) + confidenceText;
            confidenceScore.setText(lowConfidenceText);
        }
    }

        boolean success = false;

        // Try to delete by position first (more reliable)
        if (scanPosition != -1) {
            success = scanHistoryManager.deleteScanByPosition(scanPosition);
        }

        // If position method fails, try object matching
        if (!success && scanHistory != null) {
            success = scanHistoryManager.deleteScan(scanHistory);
        }

        if (success) {
            Toast.makeText(requireContext(), "Scan deleted successfully", Toast.LENGTH_SHORT).show();
            // Navigate back to previous screen
            Navigation.findNavController(requireView()).popBackStack();
        } else {
            Toast.makeText(requireContext(), "Failed to delete scan", Toast.LENGTH_SHORT).show();
        }
    }

        TextView confidenceScore = getView().findViewById(R.id.confidenceScore);

            imageResult.setImageResource(R.drawable.ic_plant_default);
        }

image utils 
package com.example.plantguard.utils;

import android.graphics.Bitmap;
import android.graphics.BitmapFactory;
import android.graphics.Matrix;
import androidx.exifinterface.media.ExifInterface;
import java.io.File;
import java.io.FileOutputStream;
import java.io.IOException;

public class ImageUtils {

    public static Bitmap rotateImageIfRequired(Bitmap bitmap, String imagePath) {
        try {
            ExifInterface exif = new ExifInterface(imagePath);
            int orientation = exif.getAttributeInt(ExifInterface.TAG_ORIENTATION, ExifInterface.ORIENTATION_NORMAL);

            Matrix matrix = new Matrix();
            switch (orientation) {
                case ExifInterface.ORIENTATION_ROTATE_90:
                    matrix.postRotate(90);
                    break;
                case ExifInterface.ORIENTATION_ROTATE_180:
                    matrix.postRotate(180);
                    break;
                case ExifInterface.ORIENTATION_ROTATE_270:
                    matrix.postRotate(270);
                    break;
                default:
                    return bitmap;
            }

            return Bitmap.createBitmap(bitmap, 0, 0, bitmap.getWidth(), bitmap.getHeight(), matrix, true);
        } catch (IOException e) {
            e.printStackTrace();
            return bitmap;
        }
    }

    public static File compressImage(File imageFile) {
        try {
            BitmapFactory.Options options = new BitmapFactory.Options();
            options.inJustDecodeBounds = false;

            Bitmap bitmap = BitmapFactory.decodeFile(imageFile.getAbsolutePath(), options);

            File compressedFile = new File(imageFile.getParent(), "compressed_" + imageFile.getName());
            FileOutputStream outputStream = new FileOutputStream(compressedFile);

            bitmap.compress(Bitmap.CompressFormat.JPEG, Constants.IMAGE_QUALITY, outputStream);
            outputStream.flush();
            outputStream.close();

            return compressedFile;
        } catch (IOException e) {
            e.printStackTrace();
            return imageFile;
        }
    }

    public static Bitmap decodeSampledBitmapFromFile(String path, int reqWidth, int reqHeight) {
        final BitmapFactory.Options options = new BitmapFactory.Options();
        options.inJustDecodeBounds = true;
        BitmapFactory.decodeFile(path, options);

        options.inSampleSize = calculateInSampleSize(options, reqWidth, reqHeight);
        options.inJustDecodeBounds = false;

        return BitmapFactory.decodeFile(path, options);
    }

    private static int calculateInSampleSize(BitmapFactory.Options options, int reqWidth, int reqHeight) {
        final int height = options.outHeight;
        final int width = options.outWidth;
        int inSampleSize = 1;

        if (height > reqHeight || width > reqWidth) {
            final int halfHeight = height / 2;
            final int halfWidth = width / 2;

            while ((halfHeight / inSampleSize) >= reqHeight && (halfWidth / inSampleSize) >= reqWidth) {
                inSampleSize *= 2;
            }
        }
        return inSampleSize;
    }
}

package com.example.plantguard.utils;

import android.content.Context;
import android.content.SharedPreferences;
import com.example.plantguard.models.ScanHistory;
import com.google.gson.Gson;
import com.google.gson.reflect.TypeToken;
import java.lang.reflect.Type;
import java.util.ArrayList;
import java.util.List;

public class ScanHistoryManager {
    private static final String PREF_NAME = "scan_history";
    private static final String KEY_SCAN_HISTORY = "scan_history_list";
    private static final int MAX_HISTORY_ITEMS = 50;

    private SharedPreferences sharedPreferences;
    private Gson gson;

    public ScanHistoryManager(Context context) {
        sharedPreferences = context.getSharedPreferences(PREF_NAME, Context.MODE_PRIVATE);
        gson = new Gson();
    }

    // Add scan to history
    public void addScanHistory(ScanHistory scan) {
        List<ScanHistory> history = getScanHistory();
        history.add(0, scan);

        if (history.size() > MAX_HISTORY_ITEMS) {
            history = history.subList(0, MAX_HISTORY_ITEMS);
        }

        saveScanHistory(history);
    }

    // Get all scan history
    public List<ScanHistory> getScanHistory() {
        String json = sharedPreferences.getString(KEY_SCAN_HISTORY, null);
        if (json == null) {
            return new ArrayList<>();
        }

        Type type = new TypeToken<List<ScanHistory>>() {}.getType();
        List<ScanHistory> history = gson.fromJson(json, type);
        return history != null ? history : new ArrayList<>();
    }

    // Get recent scans (last 5)
    public List<ScanHistory> getRecentScans() {
        List<ScanHistory> allHistory = getScanHistory();
        int count = Math.min(allHistory.size(), 5);
        return allHistory.subList(0, count);
    }

    // Clear all scan history
    public void clearScanHistory() {
        sharedPreferences.edit().remove(KEY_SCAN_HISTORY).apply();
    }

    // Delete specific scan by position
    public void deleteScan(int position) {
        List<ScanHistory> history = getScanHistory();
        if (position >= 0 && position < history.size()) {
            history.remove(position);
            saveScanHistory(history);
        }
    }

    // Delete specific scan by matching properties (fix for object reference issue)
    public boolean deleteScan(ScanHistory scanToDelete) {
        List<ScanHistory> history = getScanHistory();

        for (int i = 0; i < history.size(); i++) {
            ScanHistory scan = history.get(i);
            // Compare by multiple properties to find the matching scan
            if (scan.getDiseaseName().equals(scanToDelete.getDiseaseName()) &&
                    scan.getConfidence() == scanToDelete.getConfidence() &&
                    scan.getScanDate().equals(scanToDelete.getScanDate()) &&
                    scan.getImagePath().equals(scanToDelete.getImagePath())) {

                history.remove(i);
                saveScanHistory(history);
                return true;
            }
        }
        return false;
    }

    // Alternative: Delete by position (more reliable)
    public boolean deleteScanByPosition(int position) {
        List<ScanHistory> history = getScanHistory();
        if (position >= 0 && position < history.size()) {
            history.remove(position);
            saveScanHistory(history);
            return true;
        }
        return false;
    }

    private void saveScanHistory(List<ScanHistory> history) {
        String json = gson.toJson(history);
        sharedPreferences.edit().putString(KEY_SCAN_HISTORY, json).apply();
    }
}

package com.example.plantguard;

import android.Manifest;
import android.content.pm.PackageManager;
import android.os.Bundle;
import androidx.core.app.ActivityCompat;
import androidx.core.content.ContextCompat;
import androidx.navigation.NavController;
import androidx.navigation.NavOptions;
import androidx.navigation.fragment.NavHostFragment;
import com.example.plantguard.databinding.ActivityMainBinding;
import java.util.ArrayList;
import java.util.List;

public class MainActivity extends BaseActivity {

    public ActivityMainBinding binding;
    private NavController navController;
    private boolean shouldUpdateBottomNav = true;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        binding = ActivityMainBinding.inflate(getLayoutInflater());
        setContentView(binding.getRoot());

        checkPermissions();
        setupNavigation();
    }

    private void checkPermissions() {
        String[] permissions = {
                Manifest.permission.CAMERA,
                Manifest.permission.WRITE_EXTERNAL_STORAGE,
                Manifest.permission.READ_EXTERNAL_STORAGE
        };

        List<String> permissionsToRequest = new ArrayList<>();
        for (String permission : permissions) {
            if (ContextCompat.checkSelfPermission(this, permission) != PackageManager.PERMISSION_GRANTED) {
                permissionsToRequest.add(permission);
            }
        }

        if (!permissionsToRequest.isEmpty()) {
            ActivityCompat.requestPermissions(this,
                    permissionsToRequest.toArray(new String[0]),
                    1001);
        }
    }

    private void setupNavigation() {
        // Get NavHostFragment
        NavHostFragment navHostFragment = (NavHostFragment) getSupportFragmentManager()
                .findFragmentById(R.id.fragment_container);

        if (navHostFragment != null) {
            navController = navHostFragment.getNavController();

            // Setup bottom navigation with custom behavior
            binding.bottomNavigation.setOnItemSelectedListener(item -> {
                int itemId = item.getItemId();

                // Prevent recursive updates
                shouldUpdateBottomNav = false;

                try {
                    // Get current destination
                    int currentDestination = navController.getCurrentDestination() != null ?
                            navController.getCurrentDestination().getId() : -1;

                    if (currentDestination == itemId) {
                        // Already on this tab - pop to root
                        popToRootFragment(itemId);
                    } else {
                        // Switching to different tab - navigate with clear back stack
                        NavOptions navOptions = new NavOptions.Builder()
                                .setLaunchSingleTop(true)
                                .setPopUpTo(getRootDestinationId(itemId), false)
                                .build();

                        navController.navigate(itemId, null, navOptions);
                    }
                    return true;
                } catch (Exception e) {
                    e.printStackTrace();
                    return false;
                } finally {
                    // Re-enable bottom nav updates after a short delay
                    new android.os.Handler().postDelayed(() -> {
                        shouldUpdateBottomNav = true;
                    }, 100);
                }
            });

            // Update bottom nav selection when destination changes
            navController.addOnDestinationChangedListener((controller, destination, arguments) -> {
                if (!shouldUpdateBottomNav || destination == null) return;

                int destId = destination.getId();

                // Update bottom nav selection for top level destinations
                if (destId == R.id.homeFragment || destId == R.id.cameraFragment ||
                        destId == R.id.libraryFragment || destId == R.id.profileFragment) {
                    binding.bottomNavigation.setSelectedItemId(destId);
                }
            });

            // Set initial selection
            binding.bottomNavigation.setSelectedItemId(R.id.homeFragment);
        }
    }

    private void popToRootFragment(int fragmentId) {
        // Pop back to the root of the current tab
        if (fragmentId == R.id.homeFragment) {
            navController.popBackStack(R.id.homeFragment, false);
        } else if (fragmentId == R.id.cameraFragment) {
            navController.popBackStack(R.id.cameraFragment, false);
        } else if (fragmentId == R.id.libraryFragment) {
            navController.popBackStack(R.id.libraryFragment, false);
        } else if (fragmentId == R.id.profileFragment) {
            navController.popBackStack(R.id.profileFragment, false);
        }
    }

    private int getRootDestinationId(int itemId) {
        if (itemId == R.id.homeFragment) {
            return R.id.homeFragment;
        } else if (itemId == R.id.cameraFragment) {
            return R.id.cameraFragment;
        } else if (itemId == R.id.libraryFragment) {
            return R.id.libraryFragment;
        } else if (itemId == R.id.profileFragment) {
            return R.id.profileFragment;
        } else {
            return R.id.homeFragment;
        }
    }

    @Override
    public void onBackPressed() {
        // If we're not on a main screen, go back
        if (navController.getCurrentDestination() != null) {
            int currentId = navController.getCurrentDestination().getId();

            if (currentId == R.id.homeFragment || currentId == R.id.cameraFragment ||
                    currentId == R.id.libraryFragment || currentId == R.id.profileFragment) {
                // Exit app if back pressed on main screens
                super.onBackPressed();
            } else {
                // Go back to previous fragment
                navController.popBackStack();
            }
        } else {
            super.onBackPressed();
        }
    }

    @Override
    public void onRequestPermissionsResult(int requestCode, String[] permissions, int[] grantResults) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults);

        if (requestCode == 1001) {
            boolean allGranted = true;
            for (int result : grantResults) {
                if (result != PackageManager.PERMISSION_GRANTED) {
                    allGranted = false;
                    break;
                }
            }

            if (!allGranted) {
                // Handle permission denial if needed
            }
        }
    }
}

package com.example.plantguard;

import android.content.Intent;
import android.content.SharedPreferences;
import android.os.Bundle;
import android.os.Handler;
import androidx.appcompat.app.AppCompatActivity;

public class SplashActivity extends AppCompatActivity {

    private static final String PREFS_NAME = "PlantGuardPrefs";
    private static final String PREF_FIRST_LAUNCH = "first_launch";

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_splash);

        // Hide action bar
        if (getSupportActionBar() != null) {
            getSupportActionBar().hide();
        }

        // Check if app is launched for the first time
        SharedPreferences prefs = getSharedPreferences(PREFS_NAME, MODE_PRIVATE);
        boolean isFirstLaunch = prefs.getBoolean(PREF_FIRST_LAUNCH, true);

        // 3 seconds splash screen
        new Handler().postDelayed(() -> {
            if (isFirstLaunch) {
                // First time - mark as launched and go to permissions
                prefs.edit().putBoolean(PREF_FIRST_LAUNCH, false).apply();
                startActivity(new Intent(SplashActivity.this, MainActivity.class));
            } else {
                // Not first time - go directly to main activity
                startActivity(new Intent(SplashActivity.this, MainActivity.class));
            }
            finish();
        }, 3000);
    }
}
